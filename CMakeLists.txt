cmake_minimum_required(VERSION 3.19)

# VitaSDK check
if(NOT DEFINED ENV{VITASDK})
  message(FATAL_ERROR
    "VITASDK not defined in environment.\n"
    "Install vitaSDK and run:\n"
    "  export VITASDK=/usr/local/vitasdk\n"
    "  source \$VITASDK/host/linux64/share/vitasdk/setup-vita-sdk.sh"
  )
endif()

set(CMAKE_TOOLCHAIN_FILE
    "$ENV{VITASDK}/share/vita.toolchain.cmake"
    CACHE PATH "vitaSDK toolchain"
)

project(tmc_vita C CXX)

set(CMAKE_C_STANDARD   11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_EXTENSIONS OFF)

# Options
option(VITA_PERSISTENT_LOGS "Redirect stdout/stderr to ux0:data/tmc/tmc_log.txt" ON)

# Compilation defines (matching PC port for USA version)
add_compile_definitions(
    PC_PORT
    NON_MATCHING
    USA
    ENGLISH
    REVISION=0
    __vita__=1
)

if(VITA_PERSISTENT_LOGS)
    add_compile_definitions(VITA_PERSISTENT_LOGS)
endif()

# Include directories (matches PC port + vita layer)
include_directories(
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/libs
    ${CMAKE_SOURCE_DIR}/port
    ${CMAKE_SOURCE_DIR}/port/vita
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/build/USA          # generated by xmake
    ${CMAKE_SOURCE_DIR}/libs/ViruaPPU/include
)

# Force include vita_main_overrides.h in all files
set(VITA_FORCE_INCLUDE
    "${CMAKE_SOURCE_DIR}/port/vita/vita_main_overrides.h"
)
add_compile_options(
    -include ${VITA_FORCE_INCLUDE}
)

# Compilation flags
add_compile_options(
    -Wall
    -Wextra
    -Wno-unused-parameter
    -Wno-missing-field-initializers
    -fno-strict-aliasing
    -fwrapv
    -fno-strict-overflow
    -O1          # O0 crashes Vita in debug builds; O1 is safe
    -g
)

# ============================================================
# SDL3 via vitaSDK
# ============================================================
# vitaSDK provides SDL3 as static lib. Link manually if not found.
find_package(SDL3 QUIET)
if(NOT SDL3_FOUND)
    # Fallback: manual link (vitaSDK installs to $VITASDK/arm-vita-eabi/)
    set(SDL3_INCLUDE_DIRS "$ENV{VITASDK}/arm-vita-eabi/include/SDL3")
    set(SDL3_LIBRARIES    "$ENV{VITASDK}/arm-vita-eabi/lib/libSDL3.a")
    add_library(SDL3::SDL3 STATIC IMPORTED)
    set_target_properties(SDL3::SDL3 PROPERTIES
        IMPORTED_LOCATION "${SDL3_LIBRARIES}"
        INTERFACE_INCLUDE_DIRECTORIES "${SDL3_INCLUDE_DIRS}"
    )
endif()

# Main target
add_executable(tmc_vita
    ${PORT_SRCS}
    ${VITA_SRCS}
    ${VIRUAPPU_SRCS}
    ${ALL_GAME_SRCS}
)

# ============================================================
# Link libraries
# ============================================================
target_link_libraries(tmc_vita
    SDL3::SDL3

    # vitaSDK modules needed for SDL3
    SceDisplay_stub
    SceCtrl_stub
    SceAudio_stub
    SceSysmodule_stub
    SceGxm_stub
    SceTouch_stub
    SceMotion_stub
    SceHid_stub

    # Math
    m
)

# Generate Self + VPK
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

vita_create_self(tmc.self tmc_vita)

vita_create_vpk(tmc.vpk "TMCVITA00" tmc.self
    VERSION "01.00"
    NAME    "Zelda: Minish Cap"
    # Icon: create sce_sys/icon0.png before building
    # FILE sce_sys/icon0.png sce_sys/icon0.png
)

# Installation instructions
add_custom_command(TARGET tmc_vita POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "=== Build complete ==="
    COMMAND ${CMAKE_COMMAND} -E echo "Install tmc.vpk on Vita via VitaShell."
    COMMAND ${CMAKE_COMMAND} -E echo "Then copy baserom.gba and rom_gaps.bin to:"
    COMMAND ${CMAKE_COMMAND} -E echo "  ux0:data/tmc/"
    COMMAND ${CMAKE_COMMAND} -E echo ""
)
